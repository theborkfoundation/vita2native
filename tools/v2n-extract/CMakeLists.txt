cmake_minimum_required(VERSION 3.18)
project(v2n-extract LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../src/package" "${CMAKE_CURRENT_BINARY_DIR}/package")
add_executable(v2n-extract src/main.cpp)
target_link_libraries(v2n-extract PRIVATE v2n_extract_core)
